\section{FederatedTinyML.ino --- Key Functions}

\subsection{TinyML Initialization}
\begin{lstlisting}[language=C++, caption={TFLite Micro initialization on MKR WAN 1310.}]
void initTinyML() {
    model = tflite::GetModel(g_model);
    static tflite::MicroInterpreter static_interpreter(
        model, resolver, tensor_arena, TENSOR_ARENA_SIZE);
    interpreter = &static_interpreter;
    interpreter->AllocateTensors();
    input = interpreter->input(0);
    output = interpreter->output(0);
}
\end{lstlisting}

\subsection{Inference and Normalization}
\begin{lstlisting}[language=C++, caption={On-device feature normalization and inference.}]
void normalizeFeatures(float* features) {
    for (int i = 0; i < NUM_FEATURES; i++) {
        features[i] = (features[i] - featureMeans[i]) / featureStds[i];
    }
}

uint8_t runInference(float* features) {
    for (int i = 0; i < NUM_FEATURES; i++)
        input->data.f[i] = features[i];
    interpreter->Invoke();
    // Read predicted path loss, map to link state
    float predictedPL = output->data.f[0];
    if (predictedPL < 117.0) return LINK_STATE_GOOD;
    else if (predictedPL < 133.0) return LINK_STATE_DEGRADED;
    else return LINK_STATE_POOR;
}
\end{lstlisting}

\subsection{Federated Simulation}
\begin{lstlisting}[language=Python, caption={FedAvg aggregation in simulation.}]
# FedAvg: weighted average of client weights
total_samples = sum(client_samples)
for layer_idx in range(len(client_weights[0])):
    layer = np.zeros_like(client_weights[0][layer_idx])
    for client_idx in range(num_clients):
        weight = client_samples[client_idx] / total_samples
        layer += weight * client_weights[client_idx][layer_idx]
    new_weights.append(layer)
global_model.set_weights(new_weights)
\end{lstlisting}
